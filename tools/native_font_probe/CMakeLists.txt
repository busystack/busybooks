cmake_minimum_required(VERSION 3.14)
project(busybooks_native_font_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This probe runs on Windows; Linux uses probe_linux_typography.sh")
endif()
add_executable(busybooks_native_font_probe windows_font_probe.cpp ../../windows/runner/reading_fonts.cpp)
target_compile_features(busybooks_native_font_probe PRIVATE cxx_std_17)
target_compile_definitions(busybooks_native_font_probe PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(busybooks_native_font_probe PRIVATE /W4 /WX /EHsc)
target_link_libraries(busybooks_native_font_probe PRIVATE dwrite d2d1 windowscodecs shlwapi ole32)
